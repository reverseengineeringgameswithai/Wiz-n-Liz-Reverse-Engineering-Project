cmake_minimum_required(VERSION 3.25)
project(WizNLizPC LANGUAGES CXX)

option(WIZ_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(wiz_build_options INTERFACE)
target_compile_features(wiz_build_options INTERFACE cxx_std_20)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(wiz_build_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
endif()
if(WIZ_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer build requires GCC or Clang with installed runtimes")
    endif()
    target_compile_options(wiz_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(wiz_build_options INTERFACE -fsanitize=address,undefined)
endif()

add_library(wiz_rnc2 STATIC src/core/rnc2.cpp)
target_include_directories(wiz_rnc2 PUBLIC src)
target_link_libraries(wiz_rnc2 PUBLIC wiz_build_options)
add_executable(wiznliz-rnc2 tools/native/rnc2_main.cpp)
target_link_libraries(wiznliz-rnc2 PRIVATE wiz_rnc2)
